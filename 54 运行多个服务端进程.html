    54 运行多个服务端进程

    启用多个v2ray进程，提供不同协议的客户端入口





    一、服务端配置

    1、配置v2ray

    [root@martlet121 ~]# mv /usr/local/etc/v2ray/config.json /usr/local/etc/v2ray/config_01.json #修改配置文件名

    [root@martlet121 ~]# cp /usr/local/etc/v2ray/config_01.json /usr/local/etc/v2ray/config_02.json #多复制一个配置文件

    [root@martlet121 ~]# vim /usr/local/etc/v2ray/config_01.json #只需修改日志路径和进程监听端口即可，其他配置一样

    {
  "log": {
    "loglevel": "warning",
    "access": "/var/log/v2ray/access_01.log",               //修改日志路径
    "error": "/var/log/v2ray/error_01.log"
  },
  "inbounds": [
    {
      "port": 20001,           //服务器监听端口
      "protocol": "vmess",              //主传入协议
      "settings": {
        "clients": [
          {
            "id": "6603eb89-d26e-46b2-a626-49e18e074a6e",              //用户ID，客户端与服务器必须相同；服务端使用此ID解密客户端传来的包
            "alterId": 0                //v4.28.1版本之后必须设置为0以启用VMessAEAD
          }
        ]
      }
    }
  ],
  "outbounds": [
    {
      "protocol": "freedom",                  //主传出协议，直连的意思
      "settings": {}
    }
  ]
}

    [root@martlet121 ~]# vim /usr/local/etc/v2ray/config_02.json #只需修改日志路径和进程监听端口即可，其他配置一样

    {
  "inbounds": [
    {
      "port": 20002,                 //服务端监听端口
      "protocol": "shadowsocks",
      "settings": {
        "method": "aes-128-gcm",
        "ota": true,                //是否开启OTA
        "password": "martlet121"                    //连接服务端时认证密码
      }
    }
  ],
  "outbounds": [
    {
      "protocol": "freedom",
      "settings": {}
    }
  ]
}

    [root@martlet121 ~]# mv /etc/systemd/system/v2ray.service /etc/systemd/system/v2ray-01.service #修改服务启动文件名

    [root@martlet121 ~]# cp /etc/systemd/system/v2ray-01.service /etc/systemd/system/v2ray-02.service #创建新的启动文件

    [root@martlet121 ~]# vim /etc/systemd/system/v2ray-01.service #修改启动文件加载的配置文件

     11 ExecStart=/usr/local/bin/v2ray run -config /usr/local/etc/v2ray/config_01.json

    [root@martlet121 ~]# vim /etc/systemd/system/v2ray-02.service #修改启动文件加载的配置文件

     11 ExecStart=/usr/local/bin/v2ray run -config /usr/local/etc/v2ray/config_02.json

    [root@martlet121 ~]# systemctl daemon-reload #重启加载服务

    [root@martlet121 ~]# systemctl start v2ray-01.service #启动服务

    [root@martlet121 ~]# systemctl start v2ray-02.service #启动服务

    [root@martlet121 ~]# systemctl enable v2ray-01.service #将服务加入开机启动

    Created symlink /etc/systemd/system/multi-user.target.wants/v2ray-01.service → /etc/systemd/system/v2ray-01.service.

    [root@martlet121 ~]# systemctl enable v2ray-02.service #将服务加入开机启动

    Created symlink /etc/systemd/system/multi-user.target.wants/v2ray-02.service → /etc/systemd/system/v2ray-02.service.

    [root@martlet121 ~]# netstat -anpt | grep 20001 #查看端口监听

    tcp6 0 0 :::20001 :::* LISTEN 8053/v2ray

    [root@martlet121 ~]# netstat -anpt | grep 20002 #查看端口监听

    tcp6 0 0 :::20002 :::* LISTEN 8062/v2ray

    [root@martlet121 ~]#



    二、vmess客户端配置

    ​ 打开客户端解压后路径config.json文件，编辑完成后运行v2ray.exe文件

    {
  "inbounds": [
    {
      "port": 1080,               //本地监听端口，浏览器配置的代理端口
      "protocol": "socks",               //入口协议为SOCKS 5
      "sniffing": {                   //流量探测，根据指定的流量类型，重置所请求的目标；识别域名后应用域名路由规则、解决dns污染、可识别BT协议再处理
        "enabled": true,
        "destOverride": ["http", "tls"]
      },
      "settings": {
        "auth": "noauth"                   //socks的认证设置，noauth代表不认证，由于socks通常在客户端使用，所以这里不认证
      }
    }
  ],
  "outbounds": [
    {
      "protocol": "vmess",                  //出口协议
      "settings": {
        "vnext": [
          {
            "address": "45.76.177.113",                   //服务器地址，请修改为你自己的服务器 IP 或域名
            "port": 20001,                  //服务器端口
            "users": [
              {
                "id": "6603eb89-d26e-46b2-a626-49e18e074a6e",                      //用户ID，必须与服务器端配置相同；客户端使用此ID加密后再传给服务端
                "alterId": 0                  //此处的值也应当与服务器相同，v4.28.1版本之后必须设置为0以启用VMessAEAD
              }
            ]
          }
        ]
      }
    }
  ]
}

    三、shadowsocks客户端配置

    ​ 打开客户端解压后路径config.json文件，编辑完成后运行v2ray.exe文件

    {
  "inbounds": [
    {
      "port": 1080,               //本地监听端口，浏览器配置代理使用的端口
      "protocol": "socks",            //入口协议为SOCKS 5
      "sniffing": {
        "enabled": true,
        "destOverride": ["http", "tls"]
      },
      "settings": {
        "auth": "noauth"                 //socks的认证设置，noauth代表不认证，由于socks通常在客户端使用，所以这里不认证
      }
    }
  ],
  "outbounds": [
    {
      "protocol": "shadowsocks",
      "settings": {
        "servers": [
          {
            "address": "45.76.177.113",                 //Shadowsocks服务器地址
            "method": "aes-128-gcm",                 //Shadowsocks服务器加密方式
            "ota": true,                //是否开启OTA，true为开启
            "password": "martlet121",                  //Shadowsocks服务器密码
            "port": 20002                      //Shadowsocks服务器监听端口号
          }
        ]
      }
    }
  ]
}
