   可用于内网穿透，达到在外网利用VPS访问不具有公网IP的内网服务器

​    客户端无需认证，任意人都可访问到



角色定义

​    A：没有公网IP的内网服务器

​    B：有公网IP的云服务器

​    C：可访问公网的客户端



工作流程

​    A主动向B发起请求，建立一个连接（支持的协议VMess、Shadowsocks、Socks）

​    用户在C上向B发起请求，想访问A上的服务

​    B接受C的请求（只支持dokodemo-door协议），通过A向B建立的连接转发给A（即B反向连接了A）

一、配置A：没有公网IP的内网服务器

[root@martlet121 ~]# <mark> vim /usr/local/etc/v2ray/config.json </mark>                     #修改服务主配置文件

```json
{
  "reverse":{                          //A的反向代理设置，必须bridges对象
    "bridges":[
      {
        "tag":"bridge",                        //反向代理标签，在路由中会用到
        "domain":"home.martlet121.cn"                         //必须和B中reverse配置里一样；用于标识反向代理的流量，不必真实存在
      }
    ]
  },
  "outbounds": [
    {                      //A连接B的outbound
      "tag":"tunnel",                         //标签，在路由中会用到
      "protocol":"vmess",
      "settings":{
        "vnext":[
          {
            "address":"45.76.177.113",                      //B服务器地址
            "port":15000,
            "users":[
              {
                "id":"6603eb89-d26e-46b2-a626-49e18e074a6e",
                "alterId":0
              }
            ]
          }
        ]
      }
    },
    {                        //最终连接内网服务的outbound
      "protocol":"freedom",
      "settings":{
      },
      "tag":"out"                    //标签，在路由中会用到
    }
  ],
  "routing":{
    "rules":[
      {                       //A主动连接B的路由规则
        "type":"field",
        "inboundTag":[
          "bridge"
        ],
        "domain":[
          "full:home.martlet121.cn"
        ],
        "outboundTag":"tunnel"
      },
      {                     //反向连接访问内网服务的规则
        "type":"field",
        "inboundTag":[
          "bridge"
        ],
        "outboundTag":"out"
      }
    ]
  }
}
```

[root@martlet121 ~]# <mark> systemctl restart v2ray </mark>                   #重启服务应用配置





二、配置B：有公网IP的云服务器

[root@martlet121 ~]# <mark> vim /usr/local/etc/v2ray/config.json </mark>                       #修改服务主配置文件

```json
{
  "reverse":{                     //B的反向代理设置，必须portals对象
    "portals":[
      {
        "tag":"portal",                      //反向代理标签，在路由中会用到
        "domain":"home.martlet121.cn"                     //必须和A中reverse配置里一样；用于标识反向代理的流量，不必真实存在
      }
    ]
  },
  "inbounds": [
    {                         //接受C请求的inbound
      "tag":"external",                        //标签，在路由中会用到
      "port":80,                        //公网对外开放的80端口，用于接收外部的HTTP访问
      "protocol":"dokodemo-door",
        "settings":{
          "address":"127.0.0.1",
          "port":80,                      //转到内网监听的80端口
          "network":"tcp"
        }
    },
    {                    //接受A主动发起的请求
      "tag": "tunnel",                       //标签，在路由中会用到
      "port":15000,
      "protocol":"vmess",
      "settings":{
        "clients":[
          {
            "id":"6603eb89-d26e-46b2-a626-49e18e074a6e",
            "alterId":0
          }
        ]
      }
    }
  ],
  "routing":{
    "rules":[
      {                          //路由规则，接收C请求后发给A
        "type":"field",
        "inboundTag":[
          "external"
        ],
        "outboundTag":"portal"
      },
      {                          //路由规则，让B能够识别这是A主动发起的反向代理连接
        "type":"field",
        "inboundTag":[
          "tunnel"
        ],
        "domain":[
          "full:home.martlet121.cn"
        ],
        "outboundTag":"portal"
      }
    ]
  }
}
```

[root@martlet121 ~]# <mark> systemctl restart v2ray </mark>                   #重启服务应用配置

[root@martlet121 ~]# <mark> ss -anpt </mark>                  #查看端口监听
