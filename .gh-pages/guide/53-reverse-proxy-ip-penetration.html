内网穿透，达到在外网利用VPS访问内网中所有地址

​ 客户端要认证才能访问到（B和C之间由dokodemo-door改为VMess，C通过V2Ray连接B，B反向代理给A，相当于C通过A代理上网）

​ 后续研究下怎么结合透明代理，访问局域网内所有设备和端口 

角色定义

​ A：没有公网IP的内网服务器

​ B：有公网IP的云服务器

​ C：可访问公网的客户端

![文章配图](<./files/53 反向代理（ip穿透）/1.png>)

一、配置A：没有公网IP的内网服务器

​ 这里和端口映射一样，没有改

[root@martlet121 ~]# vim /usr/local/etc/v2ray/config.json #修改服务主配置文件

{
  "reverse":{                          //A的反向代理设置，必须bridges对象
    "bridges":[
      {
        "tag":"bridge",                        //反向代理标签，在路由中会用到
        "domain":"home.martlet121.cn"                         //必须和B中reverse配置里一样；用于标识反向代理的流量，不必真实存在
      }
    ]
  },
  "outbounds": [
    {                      //A连接B的outbound
      "tag":"tunnel",                         //标签，在路由中会用到
      "protocol":"vmess",
      "settings":{
        "vnext":[
          {
            "address":"45.76.177.113",                      //B服务器地址
            "port":15000,
            "users":[
              {
                "id":"6603eb89-d26e-46b2-a626-49e18e074a6e",
                "alterId":0
              }
            ]
          }
        ]
      }
    },
    {                        //最终连接内网服务的outbound
      "protocol":"freedom",
      "settings":{
      },
      "tag":"out"                    //标签，在路由中会用到
    }
  ],
  "routing":{
    "rules":[
      {                       //A主动连接B的路由规则
        "type":"field",
        "inboundTag":[
          "bridge"
        ],
        "domain":[
          "full:home.martlet121.cn"
        ],
        "outboundTag":"tunnel"
      },
      {                     //反向连接访问内网服务的规则
        "type":"field",
        "inboundTag":[
          "bridge"
        ],
        "outboundTag":"out"
      }
    ]
  }
}

[root@martlet121 ~]# systemctl restart v2ray #重启服务应用配置

二、配置B：有公网IP的云服务器

​ 和端口映射相比，只修改了inbound部分；并且还可将接受A和C请求的inbound合并为一个来简化配置

[root@martlet121 ~]# vim /usr/local/etc/v2ray/config.json #修改服务主配置文件

{
  "reverse":{                    //B的反向代理设置，必须portals对象
    "portals":[
      {
        "tag":"portal",                        //反向代理标签，在路由中会用到
        "domain":"home.martlet121.cn"                     //必须和A中reverse配置里一样；用于标识反向代理的流量，不必真实存在
      }
    ]
  },
  "inbounds":[
    {                     //接受C请求的inbound
      "tag":"tunnel",                   //标签，在路由中会用到
      "port":15001,
      "protocol":"vmess",
      "settings":{
        "clients":[
          {
            "id":"a26efdb8-ef34-4278-a4e6-2af32cc010aa",
            "alterId":0
          }
        ]
      }
    },
    {                     //接受A主动发起的请求
      "tag": "interconn",                      //标签，在路由中会用到
      "port":15000,
      "protocol":"vmess",
      "settings":{
        "clients":[
          {
            "id":"6603eb89-d26e-46b2-a626-49e18e074a6e",
            "alterId":0
          }
        ]
      }
    }
  ],
  "routing":{
    "rules":[
      {                     //路由规则，接收C请求后发给A
        "type":"field",
        "inboundTag":[
          "interconn"
        ],
        "outboundTag":"portal"
      },
      {                    //路由规则，让B能够识别这是A主动发起的反向代理连接
        "type":"field",
        "inboundTag":[
          "tunnel"
        ],
        "domain":[
          "full:home.martlet121.cn"                  //将指定域名的请求发给A，如果希望将全部流量发给A，这里可以不设置域名规则
        ],
        "outboundTag":"portal"
      }
    ]
  }
}

[root@martlet121 ~]# systemctl restart v2ray #重启服务应用配置

[root@martlet121 ~]# netstat -anpt #查看端口监听

三、配置C：可访问公网的客户端

​ 打开客户端解压后路径config.json文件，编辑完成后运行v2ray.exe文件（和普通客户端配置一样）

{
  "inbounds": [
    {
      "port": 1080,               //本地监听端口，浏览器配置的代理端口
      "protocol": "socks",               //入口协议为SOCKS 5
      "sniffing": {                   //流量探测，根据指定的流量类型，重置所请求的目标；识别域名后应用域名路由规则、解决dns污染、可识别BT协议再处理
        "enabled": true,
        "destOverride": ["http", "tls"]
      },
      "settings": {
        "auth": "noauth"                   //socks的认证设置，noauth代表不认证，由于socks通常在客户端使用，所以这里不认证
      }
    }
  ],
  "outbounds": [
    {
      "protocol": "vmess",                  //出口协议
      "settings": {
        "vnext": [
          {
            "address": "45.76.177.113",                   //服务器地址，请修改为你自己的服务器 IP 或域名
            "port": 15001,                  //服务器端口
            "users": [
              {
                "id": "a26efdb8-ef34-4278-a4e6-2af32cc010aa",                      //用户ID，必须与服务器端配置相同；客户端使用此ID加密后再传给服务端
                "alterId": 0                  //此处的值也应当与服务器相同，v4.28.1版本之后必须设置为0以启用VMessAEAD
              }
            ]
          }
        ]
      }
    }
  ]
}
